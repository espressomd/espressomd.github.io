147 ESPRESSO_ATTR_ALWAYS_INLINE inline void operator()(std::size_t i,
148 std::size_t j) const {
149
150 // calc distance (component-wise, avoids constructing pos1/pos2 Vector3d
151 // on the hot early-exit path; pos1/pos2 are built lazily below only
152 // where kernels actually require them)
153 auto const d = box_geo.get_mi_vector(
154 aosoa.position(i, 0), aosoa.position(i, 1), aosoa.position(i, 2),
155 aosoa.position(j, 0), aosoa.position(j, 1), aosoa.position(j, 2));
156 auto const dist_sq = d.norm2();
157
158 // Early exit if distance > maximal global cutoff
159 if (dist_sq > system_max_cutoff_sq)
160 return;
161 auto const dist = std::sqrt(dist_sq);
162 auto const &ia_params =
163 nonbonded_ias.get_ia_param(aosoa.type(i), aosoa.type(j));
164
165 ParticleForce pf{};
166
167 // Determine which data needs to be loaded based on active algorithms
168#if defined(ESPRESSO_EXCLUSIONS) or defined(ESPRESSO_THOLE)
169 bool need_particle_pointers = false;
170#ifdef ESPRESSO_EXCLUSIONS
171 need_particle_pointers |= aosoa.has_exclusion(i) or aosoa.has_exclusion(j);
172#endif
173#ifdef ESPRESSO_THOLE
174 need_particle_pointers |=
175 thole_active(ia_params, coulomb_kernel != nullptr);
176#endif
177
178 Particle const *p1_ptr = nullptr;
179 Particle const *p2_ptr = nullptr;
180 if (need_particle_pointers) {
181 p1_ptr = unique_particles.at(i);
182 p2_ptr = unique_particles.at(j);
183 }
184#endif
185
186 /***********************************************/
187 /* non-bonded pair potentials */
188 /***********************************************/
189
190 if (dist <= ia_params.max_cut) {
191#ifdef ESPRESSO_EXCLUSIONS
192 bool skip_non_bonded = false;
193 if (aosoa.has_exclusion(i) or aosoa.has_exclusion(j)) {
194 skip_non_bonded = not do_nonbonded(*p1_ptr, *p2_ptr);
195 }
196#else
197 constexpr bool skip_non_bonded = false;
198#endif
199 if (not skip_non_bonded) {
200 pf.f += calc_central_radial_force(ia_params, d, dist);
201
202 // Only call Thole force kernel if active
203#ifdef ESPRESSO_THOLE
204 if (thole_active(ia_params, coulomb_kernel != nullptr)) {
205 pf.f += thole_pair_force(*p1_ptr, *p2_ptr, ia_params, d, dist,
206 bonded_ias, coulomb_kernel);
207 }
208#endif
209 // Only call Gay-Berne force kernel if active
210#ifdef ESPRESSO_GAY_BERNE
211 if (gay_berne_active(dist, ia_params)) {
212 auto const dir1 = aosoa.get_vector_at(aosoa.director, i);
213 auto const dir2 = aosoa.get_vector_at(aosoa.director, j);
214 pf += gb_pair_force(dir1, dir2, ia_params, d, dist);
215 }
216#endif
217#ifdef ESPRESSO_DPD
218 if (dpd_active(ia_params, thermostat.thermo_switch)) {
219 auto const pos1 = aosoa.get_vector_at(aosoa.position, i);
220 auto const pos2 = aosoa.get_vector_at(aosoa.position, j);
221 auto const vel1 = aosoa.get_vector_at(aosoa.velocity, i);
222 auto const vel2 = aosoa.get_vector_at(aosoa.velocity, j);
223 auto const force = dpd_pair_force(
224 pos1, vel1, aosoa.id(i), pos2, vel2, aosoa.id(j), *thermostat.dpd,
225 box_geo, ia_params, d, dist, dist_sq);
226 pf += force;
227 }
228#endif // ESPRESSO_DPD
229
230 } // not skip_non_bonded
231 } // not dist > ia_params.max_cut
232
233 /*********************************************************************/
234 /* everything before this contributes to the virial pressure in NpT */
235 /* via d (x) pf.f; electrostatic and dipolar real-space contributions */
236 /* are added in explicitly below instead: Coulomb reuses the pair */
237 /* energy as a virial proxy, dipoles compute d . F directly (see */
238 /* rationale below) */
239 /*********************************************************************/
240#ifdef ESPRESSO_NPT
241 Utils::Vector3d virial{};
242 if (npt_active()) {
243 virial = hadamard_product(pf.f, d);
244 }
245#endif // ESPRESSO_NPT
246
247#ifdef ESPRESSO_ELECTROSTATICS
248 Utils::Vector3d f1_asym{};
249 Utils::Vector3d f2_asym{};
250 // real-space electrostatic charge-charge interaction
251 if (coulomb_kernel != nullptr) {
252 if ((aosoa.charge(i) != 0.) and (aosoa.charge(j) != 0.)) {
253 auto const q1q2 = aosoa.charge(i) * aosoa.charge(j);
254 pf.f += coulomb_pair_force(q1q2, d, dist, coulomb_kernel
255#ifdef ESPRESSO_P3M
256 ,
257 p3m
258#endif
259 );
260 if (elc_kernel) {
261 auto const pos1 = aosoa.get_vector_at(aosoa.position, i);
262 auto const pos2 = aosoa.get_vector_at(aosoa.position, j);
263 (*elc_kernel)(pos1, pos2, f1_asym, f2_asym, q1q2);
264 }
265#ifdef ESPRESSO_NPT
266 if (npt_active()) {
267 auto const pos1 = aosoa.get_vector_at(aosoa.position, i);
268 auto const pos2 = aosoa.get_vector_at(aosoa.position, j);
269 virial[0] += (*coulomb_u_kernel)(pos1, pos2, q1q2, d, dist);
270 }
271#endif // ESPRESSO_NPT
272 }
273 }
274#endif // ESPRESSO_ELECTROSTATICS
275
276#ifdef ESPRESSO_DIPOLES
277 if (dipoles_kernel != nullptr) {
278 auto const d1d2 = aosoa.dipm(i) * aosoa.dipm(j);
279 if (d1d2 != 0.) {
280 auto const dir1 = aosoa.get_vector_at(aosoa.director, i);
281 auto const dir2 = aosoa.get_vector_at(aosoa.director, j);
282 auto const dip_pf = (*dipoles_kernel)(
283 d1d2, aosoa.dipm(i) * dir1, aosoa.dipm(j) * dir2, d, dist, dist_sq);
284#ifdef ESPRESSO_NPT
285 if (npt_active()) {
286 // d . F = -n * U for a homogeneous potential of degree n
287 // (Euler's theorem, independent of centrality); n=-3 here vs
288 // n=-1 for Coulomb. Ewald screening makes that only
289 // approximate, and for dipoles the approximation measurably
290 // fails NpT pressure consistency (see test_pressure_with_dp3m),
291 // so d . F is computed explicitly here instead.
292 virial[0] += d * dip_pf.f;
293 }
294#endif // ESPRESSO_NPT
295 pf += dip_pf;
296 }
297 }
298#endif // ESPRESSO_DIPOLES
299
300 auto opf = calc_opposing_force(pf, d);
301#ifdef ESPRESSO_ELECTROSTATICS
302 pf.f += f1_asym;
303 opf.f += f2_asym;
304#endif // ESPRESSO_ELECTROSTATICS
305
306 auto access_force = local_force.access();
307
308 access_force(i, 0) += pf.f[0];
309 access_force(i, 1) += pf.f[1];
310 access_force(i, 2) += pf.f[2];
311#ifdef ESPRESSO_ROTATION
312 auto access_torque = local_torque.access();
313 access_torque(i, 0) += pf.torque[0];
314 access_torque(i, 1) += pf.torque[1];
315 access_torque(i, 2) += pf.torque[2];
316#endif
317
318 access_force(j, 0) += opf.f[0];
319 access_force(j, 1) += opf.f[1];
320 access_force(j, 2) += opf.f[2];
321#ifdef ESPRESSO_ROTATION
322 access_torque(j, 0) += opf.torque[0];
323 access_torque(j, 1) += opf.torque[1];
324 access_torque(j, 2) += opf.torque[2];
325#endif
326#ifdef ESPRESSO_NPT
327 if (npt_active()) {
328 auto access_virial = local_virial.access();
329 access_virial(0) += virial[0];
330 access_virial(1) += virial[1];
331 access_virial(2) += virial[2];
332 }
333#endif
334 }
335};

361/**
362 * @brief Own-the-loop specialization of the non-bonded pair kernel.
363 *
364 * Handles the common case of a cuboid box with only central radial pair
365 * forces active, optionally with real-space electrostatics (@p HasCoulomb).
366 * The dispatch in forces.cpp selects it only when no torque-producing,
367 * asymmetric, thermostat or exclusion interaction is active (no dipoles, ELC,
368 * DPD, NPT virial, exclusions, Thole or Gay-Berne, no Lees-Edwards), and
369 * falls back to @ref ForcesKernel otherwise.
370 *
371 * Unlike @ref ForcesKernel (invoked once per pair by
372 * Cabana::neighbor_parallel_for), this owns the Verlet-list loop and runs once
373 * per particle. That lets it hoist the per-particle position, type and charge,
374 * capture the cuboid box parameters by value, and obtain the ScatterView
375 * accessor once per particle instead of once per pair -- the per-pair
376 * `access()` is an omp_get_thread_num call that dominates the LJ pair cost. It
377 * also accumulates the i-side force in a local register over all of particle
378 * i's neighbors and updates the ScatterView once per particle instead of once
379 * per pair; the j-side write stays per-pair (each neighbor j is distinct). This
380 * removes the per-pair i-side accessor update at the cost of changing the
381 * i-side summation order (a single register sum added once, rather than
382 * incremental scatter adds), so the result matches @ref ForcesKernel to
383 * floating-point round-off rather than bitwise. `if constexpr (HasCoulomb)`
384 * compiles the electrostatics path in or out entirely.
385 *
386 * The neighbors are processed in fixed-size tiles, each in three passes: a
387 * scalar gather of the neighbor positions into SoA scratch, a vectorized
388 * minimum-image pass (@ref CuboidMinimumImage::batch_vector_dist2) that folds
389 * the whole tile at once and squares the distances, and a scalar pass that
390 * runs the short-range force only for the pairs whose squared distance passes
391 * the cutoff gate. The vectorized pass yields the same per-pair fold vector and
392 * squared distance as the scalar path, so identity is preserved.
393 */
394template <bool HasCoulomb> struct SpecializedForcesKernel {
395 static constexpr int tile_size = 64;
396
397 InteractionsNonBonded const &nonbonded_ias;
398 CellStructure::AoSoA_pack const &aosoa;
399 CellStructure::ScatterForce local_force;
400 Kokkos::View<int const *, Kokkos::HostSpace> counts;
401 Kokkos::View<int const **, Kokkos::LayoutRight, Kokkos::HostSpace> neighbors;
402 CuboidMinimumImage minimum_image;
403 double system_max_cutoff_sq;
404#ifdef ESPRESSO_ELECTROSTATICS
405 Coulomb::ShortRangeForceKernel::kernel_type const *coulomb_kernel;
406#ifdef ESPRESSO_P3M
407 CoulombP3M const *p3m;
408#endif
409#endif
410
411 ESPRESSO_ATTR_ALWAYS_INLINE KOKKOS_INLINE_FUNCTION void
412 operator()(std::size_t const i) const {
413 auto const n_neighbors = counts(i);
414 if (n_neighbors == 0)
415 return;
416
417 auto const x_i = aosoa.position(i, 0);
418 auto const y_i = aosoa.position(i, 1);
419 auto const z_i = aosoa.position(i, 2);
420 auto const type_i = aosoa.type(i);
421#ifdef ESPRESSO_ELECTROSTATICS
422 double charge_i = 0.;
423 if constexpr (HasCoulomb) {
424 charge_i = aosoa.charge(i);
425 }
426#endif
427
428 // One ScatterView accessor for all of this particle's pairs.
429 auto access_force = local_force.access();
430
431 // Accumulate the i-side force over all neighbors in a local register and
432 // write it to the ScatterView once at the end (see class docs).
433 Utils::Vector3d f_i{};
434
435 // Per-tile SoA scratch (thread-local, on the stack).
436 int js[tile_size];
437 double sx[tile_size], sy[tile_size], sz[tile_size];
438 double dx0[tile_size], dx1[tile_size], dx2[tile_size], dsq[tile_size];
439
440 for (int base = 0; base < n_neighbors; base += tile_size) {
441 auto const m = Kokkos::min(tile_size, n_neighbors - base);
442
443 // Pass 1: scalar gather of the tile's neighbor positions.
444 for (int t = 0; t < m; ++t) {
445 auto const j = neighbors(i, base + t);
446 js[t] = j;
447 auto const row_j = static_cast<std::size_t>(j);
448 sx[t] = aosoa.position(row_j, 0);
449 sy[t] = aosoa.position(row_j, 1);
450 sz[t] = aosoa.position(row_j, 2);
451 }
452
453 // Pass 2: vectorized minimum-image fold + squared distance.
454 minimum_image.batch_vector_dist2(x_i, y_i, z_i, m, sx, sy, sz, dx0, dx1,
455 dx2, dsq);
456
457 // Pass 3: scalar short-range force for the pairs that pass the gate,
458 // in neighbor order (i before j) to preserve the accumulation order.
459 for (int t = 0; t < m; ++t) {
460 if (dsq[t] > system_max_cutoff_sq)
461 continue;
462 auto const j = static_cast<std::size_t>(js[t]);
463 Utils::Vector3d const d{dx0[t], dx1[t], dx2[t]};
464 auto const dist = std::sqrt(dsq[t]);
465 auto const &ia_params =
466 nonbonded_ias.get_ia_param(type_i, aosoa.type(j));
467
468 Utils::Vector3d f{};
469 if (dist <= ia_params.max_cut) {
470 f += calc_central_radial_force(ia_params, d, dist);
471 }
472#ifdef ESPRESSO_ELECTROSTATICS
473 if constexpr (HasCoulomb) {
474 auto const charge_j = aosoa.charge(j);
475 if (charge_i != 0. and charge_j != 0.) {
476 auto const q1q2 = charge_i * charge_j;
477 f += coulomb_pair_force(q1q2, d, dist, coulomb_kernel
478#ifdef ESPRESSO_P3M
479 ,
480 p3m
481#endif
482 );
483 }
484 }
485#endif
486
487 f_i += f;
488 access_force(j, 0) -= f[0];
489 access_force(j, 1) -= f[1];
490 access_force(j, 2) -= f[2];
491 }
492 }
493
494 // Single i-side ScatterView update for the whole neighbor loop.
495 access_force(i, 0) += f_i[0];
496 access_force(i, 1) += f_i[1];
497 access_force(i, 2) += f_i[2];
498 }
499};

57 * @brief Position and dipole moment of one particle.

59struct PosMom {
60 Utils::Vector3d pos;
61 Utils::Vector3d m;
62
63 template <class Archive> void serialize(Archive &ar, long int) {
64 ar & pos & m;
65 }
66};

68static auto gather_particle_data(BoxGeometry const &box_geo,
69 ParticleRange const &particles) {
70 auto const &comm = ::comm_cart;
71 std::vector<Particle *> local_particles;
72 std::vector<PosMom> local_posmom;
73 std::vector<PosMom> all_posmom;
74 std::vector<boost::mpi::request> reqs;
75
76 local_particles.reserve(particles.size());
77 local_posmom.reserve(particles.size());
78
79 for (auto &p : particles) {
80 if (p.dipm() != 0.0) {
81 local_particles.emplace_back(&p);
82 local_posmom.emplace_back(
83 PosMom{box_geo.folded_position(p.pos()), p.calc_dip()});
84 }
85 }
86
87 auto const local_size = static_cast<int>(local_posmom.size());
88 std::vector<int> all_sizes;
89 boost::mpi::all_gather(comm, local_size, all_sizes);
90
91 auto const offset =
92 std::accumulate(all_sizes.begin(), all_sizes.begin() + comm.rank(), 0);
93 auto const total_size =
94 std::accumulate(all_sizes.begin() + comm.rank(), all_sizes.end(), offset);
95
96 if (comm.size() > 1) {
97 all_posmom.resize(total_size);
98 reqs = Utils::Mpi::iall_gatherv(comm, local_posmom.data(), local_size,
99 all_posmom.data(), all_sizes.data());
100 } else {
101 std::swap(all_posmom, local_posmom);
102 }
103
104 return std::make_tuple(std::move(local_particles), std::move(all_posmom),
105 std::move(reqs), offset);
106}

108static auto get_n_cut(BoxGeometry const &box_geo, int n_replicas) {
109 return n_replicas * Utils::Vector3i{static_cast<int>(box_geo.periodic(0)),
110 static_cast<int>(box_geo.periodic(1)),
111 static_cast<int>(box_geo.periodic(2))};
112}

114/**
115 * Real-space image shifts n x box_l inside the |ncut| sphere. Index 0 is the
116 * primary (zero) shift so self-interaction loops start at index 1.
117 */
118static std::vector<Utils::Vector3d>
119make_image_shifts(Utils::Vector3i const &ncut, Utils::Vector3d const &box_l) {
120 auto const ncut2 = ncut.norm2();
121 std::vector<Utils::Vector3d> shifts;
122 shifts.push_back({0., 0., 0.});
123 for (int nx = -ncut[0]; nx <= ncut[0]; ++nx)
124 for (int ny = -ncut[1]; ny <= ncut[1]; ++ny)
125 for (int nz = -ncut[2]; nz <= ncut[2]; ++nz) {
126 if (nx == 0 && ny == 0 && nz == 0)
127 continue;
128 if (nx * nx + ny * ny + nz * nz <= ncut2)
129 shifts.push_back({nx * box_l[0], ny * box_l[1], nz * box_l[2]});
130 }
131 return shifts;
132}

134/**
135 * @brief Calculate and add the interaction forces/torques to the particles.
136 *
137 * This employs a parallel N-square loop over all particle pairs.
138 * The computation the partitioned into several steps so that the
139 * communication latency can be hidden behind some local computation:
140 *
141 * 1. The local particle positions and momenta are packed into
142 * one array.
143 * 2. The asynchronous distribution of the local arrays to all
144 * ranks is started.
145 * 3. The interaction for the local pairs is started, here every
146 * pair is visited only once, and the force is added to both particles.
147 * 4. Wait for the data from the other nodes.
148 * 5. Calculate the interaction with the rest of the particles. Here
149 * every pair is visited twice (not necessarily on the same rank)
150 * so that no reduction of the forces is needed.
151 *
152 * Logically this is equivalent to the potential calculation
153 * in @ref DipolarDirectSum::long_range_energy, which calculates
154 * a naive N-square sum, but has better performance and scaling.
155 */
156void DipolarDirectSum::add_long_range_forces_cpu() const {
157 assert(not m_is_gpu);
158 auto const &system = get_system();
159 auto const &box_geo = *system.box_geo;
160 auto const &box_l = box_geo.length();
161 auto const particles = system.cell_structure->local_particles();
162 auto [local_particles, all_posmom, reqs, offset_signed] =
163 gather_particle_data(box_geo, particles);
164
165 /* Number of image boxes considered */
166 auto const ncut = get_n_cut(box_geo, n_replicas);
167 auto const with_replicas = (ncut.norm2() > 0);
168 auto const shifts = make_image_shifts(ncut, box_l);
169
170 auto const offset = static_cast<std::size_t>(offset_signed);
171 auto const n_local = local_particles.size();
172 auto const n_total = all_posmom.size();
173
174 auto const prefactor_local = prefactor;
175
176 /* Raw pointer to the gathered AoS data; the local slice is populated before
177 * wait_all, so Phase A may read it, and it outlives all fences. Safe to
178 * capture by value in the Kokkos [=] lambdas. */
179 auto const *pm = all_posmom.data();
180
181 using memory_space = Kokkos::HostSpace;
182 using execution_space = Kokkos::DefaultHostExecutionSpace;
183 using ForceView =
184 Kokkos::View<double *[3], Kokkos::LayoutRight, Kokkos::HostSpace>;
185 using ScatterForce =
186 Kokkos::Experimental::ScatterView<double *[3], Kokkos::LayoutRight,
187 memory_space>;
188 ForceView local_force("dds_force", n_local);
189 ForceView local_torque("dds_torque", n_local);
190 ScatterForce scatter_force(local_force);
191 ScatterForce scatter_torque(local_torque);
192
193 /* Raw pointers so the Kokkos lambdas do not capture std::vector by value. */
194 auto *local_particles_ptr = local_particles.data();
195 auto const *shifts_ptr = shifts.data();
196 auto const n_shifts = shifts.size();
197
198 /* Phase A: local pairs. Each i owns its own force/torque accumulation
199 * (written directly, unique owner, no race); the Newton's-third-law
200 * partner-j contributions go through the ScatterView with a per-lane
201 * scatter. */
202 Kokkos::RangePolicy<execution_space, Kokkos::Schedule<Kokkos::Dynamic>>
203 policy_local(std::size_t{0}, n_local);
204 policy_local.set_chunk_size(64);
205 Kokkos::parallel_for(
206 "dds_local_pairs", policy_local, [=](std::size_t const i) {
207 auto const gi = offset + i;
208 auto const &pos_i = pm[gi].pos;
209 auto const &m_i = pm[gi].m;
210 PairForce fi{};
211
212 /* (a) self-images (shifts[1..], primary excluded) */
213 for (std::size_t s = 1; s < n_shifts; ++s)
214 fi += pair_force(shifts_ptr[s], m_i, m_i);
215
216 auto force_access = scatter_force.access();
217 auto torque_access = scatter_torque.access();
218
219 /* (b) pairs with j in (gi, offset + n_local) */
220 for (auto j = gi + 1; j < offset + n_local; ++j) {
221 auto const &pos_j = pm[j].pos;
222 auto const &m_j = pm[j].m;
223 auto const d0 = with_replicas ? (pos_i - pos_j)
224 : box_geo.get_mi_vector(pos_i, pos_j);
225 auto const jl = j - offset;
226 for (std::size_t s = 0; s < n_shifts; ++s) {
227 auto const rn = d0 + shifts_ptr[s];
228 auto const pf = pair_force(rn, m_i, m_j);
229 fi.f += pf.f;
230 fi.torque += pf.torque;
231 /* Conservation of angular momentum mandates that
232 * 0 = t_i + r_ij x F_ij + t_j */
233 auto const torque_j = vector_product(pf.f, rn) - pf.torque;
234 for (int c = 0; c < 3; ++c) {
235 force_access(jl, c) -= pf.f[c];
236 torque_access(jl, c) += torque_j[c];
237 }
238 }
239 }
240 /* (d) write i's own total directly (unique owner, no race) */
241 local_particles_ptr[i]->force() += prefactor_local * fi.f;
242 local_particles_ptr[i]->torque() += prefactor_local * fi.torque;
243 });
244 Kokkos::fence();
245
246 /* Wait for remote data; the remote slices of all_posmom are now populated. */
247 boost::mpi::wait_all(reqs.begin(), reqs.end());
248
249 /* Phase B: remote pairs (red [0, offset) + black [offset + n_local,
250 * n_total)), visit-twice, no scatter; accumulate only i. */
251 Kokkos::RangePolicy<execution_space> policy_remote(std::size_t{0}, n_local);
252 Kokkos::parallel_for(
253 "dds_remote_pairs", policy_remote, [=](std::size_t const i) {
254 auto const gi = offset + i;
255 auto const &pos_i = pm[gi].pos;
256 auto const &m_i = pm[gi].m;
257 PairForce fi{};
258
259 /* Two remote ranges: red [0, offset) and black [offset + n_local,
260 * n_total). Visit-twice (each remote pair is visited once per owning
261 * rank), so only i accumulates; no scatter. */
262 std::size_t const ranges[2][2] = {{std::size_t{0}, offset},
263 {offset + n_local, n_total}};
264 for (auto const &range : ranges) {
265 auto const range_begin = range[0];
266 auto const range_end = range[1];
267 for (auto j = range_begin; j < range_end; ++j) {
268 auto const &pos_j = pm[j].pos;
269 auto const &m_j = pm[j].m;
270 auto const d0 = with_replicas ? (pos_i - pos_j)
271 : box_geo.get_mi_vector(pos_i, pos_j);
272 for (std::size_t s = 0; s < n_shifts; ++s) {
273 auto const rn = d0 + shifts_ptr[s];
274 auto const pf = pair_force(rn, m_i, m_j);
275 fi.f += pf.f;
276 fi.torque += pf.torque;
277 }
278 }
279 }
280 local_particles_ptr[i]->force() += prefactor_local * fi.f;
281 local_particles_ptr[i]->torque() += prefactor_local * fi.torque;
282 });
283 Kokkos::fence();
284
285 /* Reduce the Newton's-third-law contributions and add to particles. */
286 Kokkos::Experimental::contribute(local_force, scatter_force);
287 Kokkos::Experimental::contribute(local_torque, scatter_torque);
288 Kokkos::RangePolicy<execution_space> policy_reduce(std::size_t{0}, n_local);
289 Kokkos::parallel_for(
290 "dds_reduction", policy_reduce, [=](std::size_t const i) {
291 local_particles_ptr[i]->force() +=
292 prefactor_local * Utils::Vector3d{local_force(i, 0),
293 local_force(i, 1),
294 local_force(i, 2)};
295 local_particles_ptr[i]->torque() +=
296 prefactor_local * Utils::Vector3d{local_torque(i, 0),
297 local_torque(i, 1),
298 local_torque(i, 2)};
299 });
300 Kokkos::fence();
301
302#ifdef ESPRESSO_NPT
303 // As with DipolarP3M, the energy is not a valid
304 // substitute for the virial trace for dipole-dipole interactions, so the
305 // pressure tensor (see long_range_pressure()) is reused instead.
306 if (system.has_npt_enabled()) {
307 auto const pressure_tensor = long_range_pressure();
308 get_system().npt_add_virial_contribution(
309 pressure_tensor[0u] + pressure_tensor[4u] + pressure_tensor[8u]);
310 }
311#endif
312#ifdef ESPRESSO_DIPOLE_FIELD_TRACKING
313 if (not m_is_gpu) {
314 dipole_field_at_part_cpu();
315 }
316#endif
317}

319/**
320 * @brief Calculate the interaction potential.
321 *
322 * This employs a parallel N-square loop over all particle pairs.
323 */
324double DipolarDirectSum::long_range_energy_cpu() const {
325 assert(not m_is_gpu);
326 auto const &system = get_system();
327 auto const &box_geo = *system.box_geo;
328 auto const &box_l = box_geo.length();
329 auto const particles = system.cell_structure->local_particles();
330 auto [local_particles, all_posmom, reqs, offset_signed] =
331 gather_particle_data(box_geo, particles);
332
333 /* Number of image boxes considered */
334 auto const ncut = get_n_cut(box_geo, n_replicas);
335 auto const with_replicas = (ncut.norm2() > 0);
336 auto const shifts = make_image_shifts(ncut, box_l);
337
338 auto const offset = static_cast<std::size_t>(offset_signed);
339 auto const n_local = local_particles.size();
340 auto const n_total = all_posmom.size();
341
342 /* Raw pointer to the gathered AoS data; the local slice is populated before
343 * wait_all, so Phase A may read it, and it outlives all fences. Safe to
344 * capture by value in the Kokkos [=] lambdas. */
345 auto const *pm = all_posmom.data();
346
347 using execution_space = Kokkos::DefaultHostExecutionSpace;
348
349 /* Raw pointers so the Kokkos lambdas do not capture std::vector by value. */
350 auto const *shifts_ptr = shifts.data();
351 auto const n_shifts = shifts.size();
352
353 /* Phase A: local-upper triangular sum over j in [gi, offset + n_local),
354 * i.e. the self-image energy (shifts[1..], primary excluded) plus the pairs
355 * with j in (gi, offset + n_local). Computed from the local slice of the
356 * gathered data while the remote data is still in flight. */
357 double uA = 0.;
358 Kokkos::RangePolicy<execution_space, Kokkos::Schedule<Kokkos::Dynamic>>
359 policy_local(std::size_t{0}, n_local);
360 policy_local.set_chunk_size(64);
361 Kokkos::parallel_reduce(
362 "dds_energy_local", policy_local,
363 [=](std::size_t const i, double &u_local) {
364 auto const gi = offset + i;
365 auto const &pos_i = pm[gi].pos;
366 auto const &m_i = pm[gi].m;
367
368 /* (a) self-images (shifts[1..], primary excluded) */
369 for (std::size_t s = 1; s < n_shifts; ++s)
370 u_local += pair_potential(shifts_ptr[s], m_i, m_i);
371
372 /* (b) pairs with j in (gi, offset + n_local) */
373 for (auto j = gi + 1; j < offset + n_local; ++j) {
374 auto const &pos_j = pm[j].pos;
375 auto const &m_j = pm[j].m;
376 auto const d0 = with_replicas ? (pos_i - pos_j)
377 : box_geo.get_mi_vector(pos_i, pos_j);
378 for (std::size_t s = 0; s < n_shifts; ++s)
379 u_local += pair_potential(d0 + shifts_ptr[s], m_i, m_j);
380 }
381 },
382 uA);
383
384 /* Wait for remote data, fill the black slice. The red range [0, offset) is
385 * never summed by the energy kernel (each pair is counted once on the rank
386 * owning its lower index). */
387 boost::mpi::wait_all(reqs.begin(), reqs.end());
388
389 /* Phase B: remote-black sum over j in [offset + n_local, n_total). No self
390 * term and no primary exclusion; the range is entirely remote. */
391 double uB = 0.;
392 Kokkos::RangePolicy<execution_space> policy_remote(std::size_t{0}, n_local);
393 Kokkos::parallel_reduce(
394 "dds_energy_remote", policy_remote,
395 [=](std::size_t const i, double &u_local) {
396 auto const gi = offset + i;
397 auto const &pos_i = pm[gi].pos;
398 auto const &m_i = pm[gi].m;
399 /* sum over j in [offset + n_local, n_total) */
400 for (auto j = offset + n_local; j < n_total; ++j) {
401 auto const &pos_j = pm[j].pos;
402 auto const &m_j = pm[j].m;
403 auto const d0 = with_replicas ? (pos_i - pos_j)
404 : box_geo.get_mi_vector(pos_i, pos_j);
405 for (std::size_t s = 0; s < n_shifts; ++s)
406 u_local += pair_potential(d0 + shifts_ptr[s], m_i, m_j);
407 }
408 },
409 uB);
410
411 return prefactor * (uA + uB);
412}

414/**
415 * @brief Calculate the dipolar pressure tensor.
416 *
417 * This employs a parallel N-square loop over all particle pairs.
418 * The dipole-dipole force is not central (unlike Coulomb), so the pair
419 * contribution to the virial only accounts for the force, not the torque,
420 * matching the convention used by @ref DipolarP3M.
421 */
422Utils::Vector9d DipolarDirectSum::long_range_pressure() const {
423 if (m_is_gpu) {
424 runtimeWarningMsg() << "Pressure calculation not implemented for "
425 "DipolarDirectSum on GPU.";
426 return Utils::Vector9d{};
427 }
428
429 auto const &system = get_system();
430 auto const &box_geo = *system.box_geo;
431 auto const &box_l = box_geo.length();
432 auto const particles = system.cell_structure->local_particles();
433 auto [local_particles, all_posmom, reqs, offset_signed] =
434 gather_particle_data(box_geo, particles);
435
436 /* Number of image boxes considered */
437 auto const ncut = get_n_cut(box_geo, n_replicas);
438 auto const with_replicas = (ncut.norm2() > 0);
439 auto const shifts = make_image_shifts(ncut, box_l);
440
441 auto const offset = static_cast<std::size_t>(offset_signed);
442 auto const n_local = local_particles.size();
443 auto const n_total = all_posmom.size();
444
445 /* Raw pointer to the gathered AoS data; the local slice is populated before
446 * wait_all, so Phase A may read it, and it outlives all fences. Safe to
447 * capture by value in the Kokkos [=] lambdas. */
448 auto const *pm = all_posmom.data();
449
450 using execution_space = Kokkos::DefaultHostExecutionSpace;
451
452 /* Raw pointers so the Kokkos lambdas do not capture std::vector by value. */
453 auto const *shifts_ptr = shifts.data();
454 auto const n_shifts = shifts.size();
455
456 /* The pair contribution to the virial is r_n (x) F(r_n), reduced as a flat
457 * 9-component array. The dipole-dipole force is not central, so only the
458 * force enters the virial (not the torque). */
459
460 /* Phase A: local upper-triangular sum over j in [gi, offset + n_local) --
461 * the self-image term (shifts[1..], primary excluded) plus the pairs with
462 * j in (gi, offset + n_local). Computed while remote data is in flight. */
463 Utils::Vector9d pA{};
464 Kokkos::RangePolicy<execution_space, Kokkos::Schedule<Kokkos::Dynamic>>
465 policy_local(std::size_t{0}, n_local);
466 policy_local.set_chunk_size(64);
467 auto const join_op = [](Utils::Vector9d &acc, Utils::Vector9d const &val) {
468 acc += val;
469 };
470 auto reducerA = Reduction::make_kokkos_reducer<Utils::Vector9d>(
471 // NOLINTNEXTLINE(bugprone-exception-escape)
472 [=](std::size_t const i, Utils::Vector9d &psum) noexcept {
473 auto const gi = offset + i;
474 auto const &pos_i = pm[gi].pos;
475 auto const &m_i = pm[gi].m;
476
477 /* (a) self-images (shifts[1..], primary excluded) */
478 for (std::size_t s = 1; s < n_shifts; ++s) {
479 auto const rn = shifts_ptr[s];
480 psum += Utils::flatten(
481 Utils::tensor_product(rn, pair_force(rn, m_i, m_i).f));
482 }
483
484 /* (b) pairs with j in (gi, offset + n_local) */
485 for (auto j = gi + 1; j < offset + n_local; ++j) {
486 auto const &pos_j = pm[j].pos;
487 auto const &m_j = pm[j].m;
488 auto const d0 = with_replicas ? (pos_i - pos_j)
489 : box_geo.get_mi_vector(pos_i, pos_j);
490 for (std::size_t s = 0; s < n_shifts; ++s) {
491 auto const rn = d0 + shifts_ptr[s];
492 psum += Utils::flatten(
493 Utils::tensor_product(rn, pair_force(rn, m_i, m_j).f));
494 }
495 }
496 },
497 join_op);
498 Kokkos::parallel_reduce("dds_pressure_local", policy_local, reducerA, pA);
499
500 /* Wait for remote data, fill the black slice. The red range [0, offset) is
501 * never summed (each pair is counted once on the rank owning its lower
502 * index). */
503 boost::mpi::wait_all(reqs.begin(), reqs.end());
504
505 /* Phase B: remote-black sum over j in [offset + n_local, n_total). No self
506 * term and no primary exclusion -- the range is entirely remote. */
507 Utils::Vector9d pB{};
508 Kokkos::RangePolicy<execution_space> policy_remote(std::size_t{0}, n_local);
509 auto reducerB = Reduction::make_kokkos_reducer<Utils::Vector9d>(
510 // NOLINTNEXTLINE(bugprone-exception-escape)
511 [=](std::size_t const i, Utils::Vector9d &psum) noexcept {
512 auto const gi = offset + i;
513 auto const &pos_i = pm[gi].pos;
514 auto const &m_i = pm[gi].m;
515 for (auto j = offset + n_local; j < n_total; ++j) {
516 auto const &pos_j = pm[j].pos;
517 auto const &m_j = pm[j].m;
518 auto const d0 = with_replicas ? (pos_i - pos_j)
519 : box_geo.get_mi_vector(pos_i, pos_j);
520 for (std::size_t s = 0; s < n_shifts; ++s) {
521 auto const rn = d0 + shifts_ptr[s];
522 psum += Utils::flatten(
523 Utils::tensor_product(rn, pair_force(rn, m_i, m_j).f));
524 }
525 }
526 },
527 join_op);
528 Kokkos::parallel_reduce("dds_pressure_remote", policy_local, reducerB, pB);
529
530 return prefactor * (pA + pB);
531}

533/**
534 * @brief Calculate total dipole field of each particle.
535 *
536 * This employs a parallel N-square loop over all particles.
537 * Logically this is equivalent to the potential calculation
538 * in @ref DipolarDirectSum::long_range_energy, which calculates
539 * a naive N-square sum. The difference is summation range,
540 * and the kernel calculates the dipole field rather than the energy.
541 */
542#ifdef ESPRESSO_DIPOLE_FIELD_TRACKING
543void DipolarDirectSum::dipole_field_at_part_cpu() const {
544 assert(not m_is_gpu);
545 auto const &system = get_system();
546 auto const &box_geo = *system.box_geo;
547 auto const &box_l = box_geo.length();
548 auto const particles = system.cell_structure->local_particles();
549 /* collect particle data */
550 auto [local_particles, all_posmom, reqs, offset_signed] =
551 gather_particle_data(box_geo, particles);
552
553 auto const ncut = get_n_cut(box_geo, n_replicas);
554 auto const with_replicas = (ncut.norm2() > 0);
555 auto const shifts = make_image_shifts(ncut, box_l);
556
557 auto const offset = static_cast<std::size_t>(offset_signed);
558 auto const n_local = local_particles.size();
559 auto const n_total = all_posmom.size();
560
561 auto const prefactor_local = prefactor;
562
563 /* The field sweeps over all j, so every view slice is needed. Unlike the
564 * force/energy kernels there is no local-only computation to overlap with,
565 * so wait for the remote data first, then fill all slices. */
566 boost::mpi::wait_all(reqs.begin(), reqs.end());
567
568 /* Raw pointer to the gathered AoS data; all slices are now populated, and it
569 * outlives the fence. Safe to capture by value in the Kokkos [=] lambda. */
570 auto const *pm = all_posmom.data();
571
572 using execution_space = Kokkos::DefaultHostExecutionSpace;
573
574 /* Raw pointers so the Kokkos lambdas do not capture std::vector by value. */
575 auto *local_particles_ptr = local_particles.data();
576 auto const *shifts_ptr = shifts.data();
577 auto const n_shifts = shifts.size();
578
579 Kokkos::RangePolicy<execution_space> policy(std::size_t{0}, n_local);
580 Kokkos::parallel_for("dds_dipole_field", policy, [=](std::size_t const i) {
581 auto const gi = offset + i;
582 auto const &pos_i = pm[gi].pos;
583 auto const &m_i = pm[gi].m;
584 Utils::Vector3d u{};
585
586 /* (a) self-image term over shifts[1..] (primary excluded) */
587 for (std::size_t s = 1; s < n_shifts; ++s)
588 u += dipole_field(shifts_ptr[s], m_i);
589
590 /* Sweep over all j in [0, n_total), self-primary excluded by splitting
591 * the range into [0, gi) and [gi + 1, n_total). */
592 std::size_t const ranges[2][2] = {{std::size_t{0}, gi}, {gi + 1, n_total}};
593 for (auto const &range : ranges) {
594 auto const range_begin = range[0];
595 auto const range_end = range[1];
596 for (auto j = range_begin; j < range_end; ++j) {
597 auto const &pos_j = pm[j].pos;
598 auto const &m_j = pm[j].m;
599 auto const d0 = with_replicas ? (pos_i - pos_j)
600 : box_geo.get_mi_vector(pos_i, pos_j);
601 for (std::size_t s = 0; s < n_shifts; ++s)
602 u += dipole_field(d0 + shifts_ptr[s], m_j);
603 }
604 }
605 local_particles_ptr[i]->dip_fld() = prefactor_local * u;
606 });
607 Kokkos::fence();
608}
